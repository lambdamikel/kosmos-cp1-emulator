# Counter for the Kosmos CP1: counts 0, 1, 2, ... on the display, about four steps a second.
# Cell 010 holds the count.

000 ako 04.000   # start at 0
001 abs 06.010   # count := Akku
002 lda 05.010   # Akku := count
003 anz 02.000   # show it
004 vzg 03.250   # wait 250 ms
005 ako 04.001
006 add 07.010   # Akku := 1 + count
007 abs 06.010
008 spu 09.002   # and again
